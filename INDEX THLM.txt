                const statusColors = {
                    completed: 'text-green-400',
                    current: 'text-blue-400',
                    pending: 'text-gray-500'
                };
                
                const statusIcons = {
                    completed: 'check-circle',
                    current: 'play-circle',
                    pending: 'lock'
                };
                
                const div = document.createElement('div');
                div.className = 'flex items-center gap-4 p-4 rounded-xl bg-white/5 mb-3 hover:bg-white/10 transition';
                div.innerHTML = `
                    <div class="w-10 h-10 rounded-full bg-gray-800 flex items-center justify-center ${statusColors[video.status]}">
                        <i data-lucide="${statusIcons[video.status]}" class="w-5 h-5"></i>
                    </div>
                    <div class="flex-1">
                        <h4 class="font-bold ${video.status === 'pending' ? 'text-gray-500' : 'text-white'}">
                            ${index + 1}. ${video.title}
                        </h4>
                        <p class="text-sm text-gray-400">${video.duration}</p>
                    </div>
                    ${video.status !== 'pending' ? `
                        <button class="px-4 py-2 rounded-lg bg-blue-600/20 text-blue-400 hover:bg-blue-600/30 transition text-sm">
                            مشاهدة
                        </button>
                    ` : ''}
                `;
                list.appendChild(div);
            });
            
            lucide.createIcons();
            modal.classList.remove('hidden');
        }
        
        function closeModal() {
            document.getElementById('videoModal').classList.add('hidden');
        }

            if (e.target === this) closeModal();
        });
